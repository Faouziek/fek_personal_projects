cmake_minimum_required(VERSION 3.16)
project(JeuCartes VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(cartes
    src/carte/Carte.cpp src/plateau/Plateau.cpp src/vue/IHM.cpp
    src/jeu/Jeu.cpp src/jeu/JeuUNO.cpp src/jeu/Jeu7Fam.cpp
    src/jeu/JeuPoker.cpp src/jeu/JeuTarot.cpp)
target_include_directories(cartes PUBLIC include)
if(MSVC)
    target_compile_options(cartes PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(cartes PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(jeuCartes src/Main.cpp)
target_link_libraries(jeuCartes PRIVATE cartes)
include(CTest)
if(BUILD_TESTING)
    add_executable(tests_cartes tests/TestProjet.cpp)
    target_link_libraries(tests_cartes PRIVATE cartes)
    add_test(NAME regles_et_simulations COMMAND tests_cartes)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME interface_console COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:jeuCartes>)
    endif()
endif()
